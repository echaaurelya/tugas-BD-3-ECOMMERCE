        <script>
            // fade-in on load/scroll
            const fades = document.querySelectorAll('.fade');
            function reveal() {
                fades.forEach(el => {
                    const r = el.getBoundingClientRect();
                    if (r.top < window.innerHeight - 60) el.classList.add('visible');
                });
            }
            window.addEventListener('load', reveal);
            window.addEventListener('scroll', reveal);

            // modal control
            const modalBackdrop = document.getElementById('modalBackdrop');
            const modalBox = document.getElementById('modalBox');

            function openModal(productName) {
                document.getElementById('modalProduct').value = productName;
                modalBackdrop.classList.add('show');
                modalBackdrop.setAttribute('aria-hidden', 'false');
                modalBox.classList.add('show');
                document.body.style.overflow = 'hidden';
                // focus email
                setTimeout(() => document.getElementById('email').focus(), 200);
            }
            function closeModal() {
                modalBackdrop.classList.remove('show');
                modalBackdrop.setAttribute('aria-hidden', 'true');
                modalBox.classList.remove('show');
                document.body.style.overflow = '';
            }

            function submitForm() {
                const email = document.getElementById('email').value;
                const product = document.getElementById('modalProduct').value || 'produk';
                alert('Terima kasih, ' + email + ' — pesanan untuk: ' + product + ' akan diproses. (Demo)');
                closeModal();
            }

            function openDetail(name) {
                alert('Detail singkat: ' + name + '\nLihat spesifikasi pada card produk.');
            }
